# 食材調達完了 (PP)
country_event = {
	id = OCT_decisions.1
	title = OCT_decisions.1.t
	desc = OCT_decisions.1.d
	picture = GFX_report_event_generic_read_write
	
	is_triggered_only = yes
	
	option = {
		name = OCT_decisions.1.a # 成果あり (+1)
		trigger = {
			NOT = { has_country_flag = oct_procure_food_great_success }
		}
		add_to_variable = { var = oct_food_stock value = 1 }
	}
	
	option = {
		name = OCT_decisions.1.b # 大漁 (+2)
		trigger = {
			has_country_flag = oct_procure_food_great_success
		}
		add_to_variable = { var = oct_food_stock value = 2 }
		clr_country_flag = oct_procure_food_great_success
	}
}
